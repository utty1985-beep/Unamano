  function ctPad(n){return String(n).padStart(2,'0')}
  function ctLocalDate(d){return d.getFullYear()+'-'+ctPad(d.getMonth()+1)+'-'+ctPad(d.getDate())}
  function ctHour(v){try{return new Date(v).toLocaleTimeString('it-IT',{hour:'2-digit',minute:'2-digit'})}catch{return '-'}}
  function ctAngle(a,b){return Math.abs((((+a||0)-(+b||0)+540)%360)-180)}
  function ctWindLocal(name){return String(name||'')==='Libeccio'?'Libeccio (Favugna)':String(name||'')}
  function ctBaseWind(p){
    let deg=Number.isFinite(+p.windDeg)?+p.windDeg:Number.isFinite(+p.weather?.windDeg)?+p.weather.windDeg:null;
    if(deg==null){
      const names=['Tramontana','Grecale','Levante','Scirocco','Ostro','Libeccio','Ponente','Maestrale'];
      const n=String(p.windName||p.weather?.windName||'').replace(' (Favugna)','');
      const ix=names.indexOf(n);if(ix>=0)deg=ix*45;
    }
    const speed=Number.isFinite(+p.windSpeed)?+p.windSpeed:Number.isFinite(+p.weather?.windSpeed)?+p.weather.windSpeed:null;
    return {deg:deg,speed:speed};
  }
  function ctWet(code){return [51,53,55,61,63,65,80,81,82,95,96,99].includes(+code)}
  function ctSimilarity(p,h,i){
    const base=ctBaseWind(p);
    if(base.deg==null||base.speed==null)return {good:false,missing:true,reason:'vento storico non registrato'};
    const dir=+h.wind_direction_10m?.[i]||0,speed=+h.wind_speed_10m?.[i]||0,rain=+h.precipitation_probability?.[i]||0,code=h.weather_code?.[i];
    const dd=ctAngle(base.deg,dir),ds=Math.abs(base.speed-speed);
    let good=dd<=45&&ds<=10&&rain<=55;
    const oldCond=String(p.weather?.condition||'').toLowerCase();
    if(oldCond){
      const oldWet=/piogg|rovesc|tempor|piovig/.test(oldCond),newWet=ctWet(code);
      if(oldWet!==newWet&&(oldWet||newWet))good=false;
    }
    return {good:good,missing:false,dd:dd,ds:ds,dir:dir,speed:speed,rain:rain,code:code};
  }
  function ctNearest(times,target){
    let best=-1,bd=Infinity;
    (times||[]).forEach(function(t,i){const d=Math.abs(new Date(t).getTime()-target.getTime());if(d<bd){bd=d;best=i}});
    return best;
  }
  async function ctCheckSighting(p,target){
    const day=ctLocalDate(target),next=new Date(target);next.setDate(next.getDate()+1);
    const u='https://api.open-meteo.com/v1/forecast?latitude='+encodeURIComponent(p.lat)+'&longitude='+encodeURIComponent(p.lng)+'&hourly=precipitation_probability,wind_speed_10m,wind_direction_10m,wind_gusts_10m,weather_code&wind_speed_unit=kmh&timezone=auto&start_date='+day+'&end_date='+ctLocalDate(next);
    const r=await fetch(u,{cache:'no-store'});if(!r.ok)throw Error('Meteo non disponibile');
    const j=await r.json(),h=j.hourly||{},idx=ctNearest(h.time,target);if(idx<0)throw Error('Orario non disponibile');
    const first=ctSimilarity(p,h,idx);
    if(first.missing)return {missing:true,good:false,reason:first.reason};
    let until=null,atLeast=false;
    if(first.good){
      const limit=Math.min((h.time||[]).length,idx+13);
      let jx=idx+1;
      for(;jx<limit;jx++){if(!ctSimilarity(p,h,jx).good)break}
      if(jx<limit)until=h.time[jx];
      else{until=h.time[Math.max(idx,limit-1)]||h.time[idx];atLeast=true}
    }
    return {good:first.good,missing:false,until:until,atLeast:atLeast,dir:first.dir,speed:first.speed,rain:first.rain,code:first.code,dd:first.dd,ds:first.ds};
  }

  async function runSightingChecks(){
    const date=$('ctSightingDate')?.value,time=$('ctSightingTime')?.value||'08:00';
    if(!date)return toast('Scegli il giorno di partenza');
    if(!navigator.onLine)return toast('Serve Internet per confrontare le previsioni');
