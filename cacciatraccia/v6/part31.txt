function mode(){try{return window.PFCTerritory?.getMode?.()||'hunt'}catch{return 'hunt'}}
function num(v){return v!==null&&v!==undefined&&v!==''&&Number.isFinite(+v)?+v:null}
function diff(a,b){let d=Math.abs((a-b)%360);return d>180?360-d:d}
function ref(p){
 let d=num(p.windDeg);if(d==null)d=num(p.weather?.windDeg);
 if(d==null){const names=['Tramontana','Grecale','Levante','Scirocco','Ostro','Libeccio','Ponente','Maestrale'];const i=names.indexOf(p.windName||p.weather?.windName);if(i>=0)d=i*45}
 let s=num(p.windSpeed);if(s==null)s=num(p.weather?.windSpeed);
 return {d,s};
}
function candidates(){
 return (state?.points||[]).filter(p=>{const t=String(p?.type||'').toLowerCase(),k=String(p?.sightingKind||'').toLowerCase(),s=String(p?.species||p?.name||'').toLowerCase();return Number.isFinite(+p.lat)&&Number.isFinite(+p.lng)&&p.type!=='Avvistamento'&&p.type!=='Parcheggio'&&k!=='mushroom'&&!t.includes('fung')&&!s.includes('fung')&&!s.includes('porcin')}).filter(p=>{const r=ref(p);return r.d!=null&&r.s!=null}).slice(0,8);
}
async function now(p){
 const u='https://api.open-meteo.com/v1/forecast?latitude='+p.lat+'&longitude='+p.lng+'&current=wind_speed_10m,wind_direction_10m,wind_gusts_10m&hourly=precipitation_probability&wind_speed_unit=kmh&timezone=auto&forecast_days=1';
 const rr=await fetch(u,{cache:'no-store'});if(!rr.ok)throw Error('meteo');const j=await rr.json(),x=j.current||{},h=j.hourly||{};
 let rain=0;if(Array.isArray(h.time)&&h.time.length){const target=new Date(x.time||Date.now()).getTime();let bi=0,bd=Infinity;h.time.forEach((t,i)=>{const d=Math.abs(new Date(t).getTime()-target);if(d<bd){bd=d;bi=i}});rain=+h.precipitation_probability?.[bi]||0}
 return {d:+x.wind_direction_10m||0,s:+x.wind_speed_10m||0,g:+x.wind_gusts_10m||0,r:rain};
}
function score(a,b){
 const sd=Math.max(0,1-diff(a.d,b.d)/90),ss=Math.max(0,1-Math.abs(a.s-b.s)/20),sr=Math.max(0,1-Math.max(0,b.r-20)/80);
 return Math.round((sd*.5+ss*.35+sr*.15)*100);
}
function klass(n){return n>=75?'good':n>=55?'warn':'bad'}

}

async function refresh(force=false){
 ensure();
 if(mode()!=='hunt'){renderMush();return}
 if(!navigator.onLine){const b=document.getElementById('pfcSmart');if(b)b.innerHTML='<div class="pfcSmartHead"><div><h3>🎯 Miglior punto adesso</h3><small>Serve rete per aggiornare il confronto</small></div></div>'+prep();return}
 if(!force&&cache.rows.length&&Date.now()-cache.at<TTL){renderRows(cache.rows);return}
 if(busy)return;busy=true;
 const pts=candidates(),rows=[];
 for(const p of pts){try{const w=await now(p),sc=score(ref(p),w);rows.push({p,w,sc})}catch{}}
 rows.sort((a,b)=>b.sc-a.sc);cache={at:Date.now(),rows};busy=false;renderRows(rows);
}
