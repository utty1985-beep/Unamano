  function parse(x){
    const seen=new Set();
    function w(v){
      if(v==null)return null;
      if(typeof v==='string'&&v.length>2)return {label:v,confidence:null};
      if(typeof v!=='object'||seen.has(v))return null;seen.add(v);
      if(typeof v.label==='string'){
        let c=Number(v.confidence);
        if(!Number.isFinite(c)&&Array.isArray(v.confidences)&&v.confidences[0])c=Number(v.confidences[0].confidence);
        return {label:v.label,confidence:Number.isFinite(c)?c:null};
      }
      if(Array.isArray(v.confidences)&&v.confidences.length){
        const a=[...v.confidences].sort((a,b)=>(+b.confidence||0)-(+a.confidence||0))[0];
        if(a&&a.label)return {label:a.label,confidence:Number(a.confidence)||null};
      }
      const nums=Object.entries(v).filter(e=>typeof e[1]==='number'&&Number.isFinite(e[1]));
      if(nums.length>=2){nums.sort((a,b)=>b[1]-a[1]);return {label:nums[0][0],confidence:nums[0][1]}}
      if(Array.isArray(v)){for(const z of v){const r=w(z);if(r)return r}}
      else{
        for(const k of ['data','value','prediction','result','output'])if(k in v){const r=w(v[k]);if(r)return r}
        for(const z of Object.values(v)){const r=w(z);if(r)return r}
      }
      return null;
    }
    return w(x);
  }
